# ====================================================================================================
# COMANDOS PARA INSTALAÇÃO DO MÓDULO DE BOLETOS - CAIXA ECONÔMICA FEDERAL
# ====================================================================================================
# Execute estes comandos na ordem para instalar o módulo completo de boletos
# ====================================================================================================

# 1. Criar migrations do módulo boletos
python manage.py makemigrations boletos

# 2. Aplicar migrations no banco de dados
python manage.py migrate boletos

# 3. Instalar dependências Python necessárias
pip install reportlab python-barcode pillow

# 4. Criar módulo, menus e permissões no sistema
python criar_modulo_boletos.py

# 5. (Opcional) Atualizar requirements.txt com as novas dependências
pip freeze > requirements.txt

# ====================================================================================================
# APÓS EXECUTAR OS COMANDOS ACIMA:
# ====================================================================================================
# 
# 1. Acesse o Admin Django: http://127.0.0.1:8000/admin/
# 2. Vá em "Boletos" → "Configurações Bancárias"  
# 3. Clique em "Adicionar Configuração Bancária"
# 4. Preencha com os dados fornecidos pela Caixa Econômica Federal
# 5. Marque como "Ativo"
# 6. Salve
# 
# 7. Acesse o módulo: http://127.0.0.1:8000/boletos/
# 8. Teste gerando um boleto para uma parcela
